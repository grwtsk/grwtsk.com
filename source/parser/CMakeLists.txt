cmake_minimum_required(VERSION 3.20)
project(correspondence_parser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(CMARK_INCLUDE NAMES cmark-gfm.h PATH_SUFFIXES cmark-gfm REQUIRED)
find_path(CMARK_EXT_INCLUDE NAMES cmark-gfm-core-extensions.h PATH_SUFFIXES cmark-gfm REQUIRED)
find_library(CMARK NAMES cmark-gfm REQUIRED)
find_library(CMARK_EXT NAMES cmark-gfm-extensions REQUIRED)
add_executable(markdown-parser src/main.cpp)
target_include_directories(markdown-parser PRIVATE ${CMARK_INCLUDE} ${CMARK_EXT_INCLUDE})
if(TARGET yaml-cpp::yaml-cpp)
  set(YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(YAML_TARGET yaml-cpp)
endif()
target_link_libraries(markdown-parser PRIVATE ${YAML_TARGET} nlohmann_json::nlohmann_json ${CMARK_EXT} ${CMARK})
target_compile_options(markdown-parser PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic>)
